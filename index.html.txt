    <script>
        // Server Location & Theme State Manager
        let currentServer = {
            name: "Richmond Metropolitan",
            acronym: "RM",
            subtitle: "Staff Portal • ERLC"
        };

        function handleLocationChange(val) {
            const customBox = document.getElementById('custom-location-box');
            if (val === 'Custom') {
                customBox.classList.remove('hidden');
                applyCustomLocation();
            } else {
                customBox.classList.add('hidden');
                let acronym = "ERLC";
                if (val.includes("Richmond")) acronym = "RM";
                else if (val.includes("Liberty")) acronym = "LC";
                else if (val.includes("Angeles")) acronym = "LAPD";
                else if (val.includes("Miami")) acronym = "MDPD";
                else if (val.includes("New York")) acronym = "NYPD";
                else if (val.includes("London")) acronym = "MPS";

                updateServerLocation(val, acronym);
            }
        }

        function applyCustomLocation() {
            const customName = document.getElementById('custom-server-input').value.trim() || "Custom Roleplay";
            const customAcronym = document.getElementById('custom-acronym-input').value.trim() || "RP";
            updateServerLocation(customName, customAcronym);
        }

        function updateServerLocation(name, acronym) {
            currentServer.name = name;
            currentServer.acronym = acronym;

            // Update UI elements dynamically
            document.getElementById('server-brand-title').textContent = name.toUpperCase();
            document.getElementById('server-brand-subtitle').textContent = acronym + " Staff Portal";
            document.getElementById('server-brand-icon').textContent = acronym.charAt(0);
            
            document.getElementById('header-server-title').textContent = name + " Staff Portal";
            document.getElementById('welcome-server-title').textContent = name + " Moderation Hub";

            showToast("Server location set to: " + name);
            generateMarkdown();
        }

        // Tab Switcher Logic
        function switchTab(tabId) {
            const contents = document.querySelectorAll('.tab-content');
            contents.forEach(el => el.classList.add('hidden'));

            const buttons = document.querySelectorAll('.nav-btn');
            buttons.forEach(btn => {
                btn.classList.remove('active', 'text-cyan-400', 'bg-cyan-500/10', 'border', 'border-cyan-500/30', 'text-emerald-400', 'bg-emerald-500/20', 'border-emerald-500/30');
                btn.classList.add('text-slate-400');
            });

            const targetContent = document.getElementById('tab-' + tabId);
            if (targetContent) {
                targetContent.classList.remove('hidden');
            }

            const targetBtn = document.getElementById('nav-' + tabId);
            if (targetBtn) {
                if (tabId === 'github-guide') {
                    targetBtn.classList.add('active', 'text-emerald-400', 'bg-emerald-500/20', 'border', 'border-emerald-500/30');
                } else {
                    targetBtn.classList.add('active', 'text-cyan-400', 'bg-cyan-500/10', 'border', 'border-cyan-500/30');
                }
                targetBtn.classList.remove('text-slate-400');
            }
        }

        // Search Filter
        function filterSopRules() {
            const input = document.getElementById('sop-search').value.toLowerCase();
            const cards = document.querySelectorAll('.sop-card');

            cards.forEach(card => {
                const text = card.innerText.toLowerCase();
                card.style.display = text.includes(input) ? "block" : "none";
            });
        }

        // Custom Toast Helper (Replaces alert)
        function showToast(message) {
            const toast = document.getElementById('toast');
            const toastMsg = document.getElementById('toast-message');
            toastMsg.textContent = message;
            toast.classList.remove('translate-y-20', 'opacity-0');
            toast.classList.add('translate-y-0', 'opacity-100');

            setTimeout(() => {
                toast.classList.remove('translate-y-0', 'opacity-100');
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 2500);
        }

        // Clipboard Helper
        function copyToClipboard(text) {
            const tempInput = document.createElement('textarea');
            tempInput.value = text;
            document.body.appendChild(tempInput);
            tempInput.select();
            document.execCommand('copy');
            document.body.removeChild(tempInput);

            showToast("Copied '" + text + "' to clipboard!");
        }

        // Markdown Generator
        function generateMarkdown() {
            const discord = document.getElementById('gen-discord').value.trim() || '@User';
            const roblox = document.getElementById('gen-roblox').value.trim() || 'N/A';
            const action = document.getElementById('gen-action').value;
            const rule = document.getElementById('gen-rule').value;
            const proof = document.getElementById('gen-proof').value.trim() || 'No proof linked';
            const notes = document.getElementById('gen-notes').value.trim() || 'None provided.';

            const header = "[ " + currentServer.name.toUpperCase() + " INCIDENT LOG ]";
            const markdown = "```ansi\n" +
                "\u001b[1;36m" + header + "\u001b[0m\n" +
                "> **Target Member:** " + discord + "\n" +
                "> **Roblox Username:** `" + roblox + "`\n" +
                "> **Action Issued:** **" + action + "**\n" +
                "> **Rule Violated:** " + rule + "\n" +
                "> **Evidence:** " + proof + "\n" +
                "> **Moderator Notes:** " + notes + "\n" +
                "> **Logged By:** Staff Member On Duty\n" +
                "> **Timestamp:** " + new Date().toLocaleString() + "\n" +
                "```";

            document.getElementById('markdown-output').textContent = markdown;
        }

        function copyOutput() {
            const text = document.getElementById('markdown-output').textContent;
            copyToClipboard(text);
        }

        window.onload = function() {
            generateMarkdown();
        };
    </script>
